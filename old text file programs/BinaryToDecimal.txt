#include <iostream>
using namespace std;

int BinToDec(int binNum){
    int ans = 0;
    int power = 1;
    while( binNum > 0 ){
        int rem = binNum % 10;
        binNum /= 10;
        ans += (rem*power);
        power *= 2;
    }
    return ans;
}

int main()
{
    int x;
    cout << "enter a binary number :";
    cin >> x;
    cout << "(" << x << ")2 =" << "(" << BinToDec(x) << ")10";
    return 0;
}
