#include <iostream>
using namespace std;

int multi(int arr[],int size){
    int mul = 1;
    for(int i = 0; i < size; i++){
        mul *= arr[i];
    }
    return mul;
}

int sum(int arr[],int size){
    int sum = 0;
    for(int i = 0; i < size; i++){
        sum += arr[i];
    }
    return sum;
}

int main()
{
    int arr[] = { 1, 2, 3, 4, 5};
    int size = 5;
    cout << "sum of array = " << sum(arr,size) << endl;
    cout << "multiplication of array = " << multi(arr,size) << endl;
    
    return 0;
}
