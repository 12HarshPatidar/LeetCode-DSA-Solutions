#include <iostream>
using namespace std;

int fab(int x){
    int a = 0;
    int b = 1;
    cout << a << endl << b << endl;
    for(int i = 0; i < x; i++){
        int c = a+b;
        cout << c << endl;
        a = b;
        b = c;
    }
    return 0;
}

int main()
{
    int n;
    cout << "enter the nth term :" ; 
    cin >> n;
    fab(n);
    return 0;
}
