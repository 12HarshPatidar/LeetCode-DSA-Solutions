#include <iostream>
using namespace std;

int factorial(int x){
    int fact = 1;
    for(int i = 1;i <= x; i++){
        fact = fact*i;
    }
    return fact;
}
int main()
{
    int n,r;
    cout << "enter the value of n :";
    cin >> n;
    cout << "enter the value of r :";
    cin >> r;
    
    int fact_n = factorial(n);
    int fact_r = factorial(r);
    int fact_nr = factorial(n-r);
    
    int ncr = fact_n/(fact_r*fact_nr);
    cout << "nCr = " << ncr;
    return 0;
}
