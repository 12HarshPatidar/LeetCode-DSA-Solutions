#include <iostream>
using namespace std;

void unique(int arr[],int size){
    
    for(int i = 0; i < size; i++){
        bool che = true;
        for(int j = 0; j < size; j++){
            if(i != j){
                if(arr[i] == arr[j]){
                    che = false;
                }
            }
        }
    if(che == true){
        cout << arr[i] << " "; 
    }
    }
}

int main()
{
   int arr[] = { 1, 2 , 3, 4, 5, 6, 2, 3, 5};
   int size = sizeof(arr)/sizeof(int);
   unique(arr,size);
   
   return 0;
}
