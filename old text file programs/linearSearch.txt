#include <iostream>
using namespace std;

int search(int arr[],int target){
    for(int i = 0; i < 8; i++){
        if(arr[i] == target){
            return i;
        }
    }
    return -1;
}

int main()
{
    int arr[]={1,2,3,4,5,6,7,8};
    int target = 2;
    cout << "Target is at " << search(arr,target) << endl;
    
    return 0;
}
