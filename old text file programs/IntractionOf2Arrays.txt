#include <iostream>
using namespace std;

int intraction(int arr1[],int size1,int arr2[],int size2){
    int arr3[100];
    int k = 0;
    for(int i = 0; i < size1; i++){
        for(int j = 0; j < size2; j++){
            if(arr1[i] == arr2[j]){
                arr3[k++] = arr1[i];
                break;
            }
        }
    }
    for(int i = 0; i < k; i++){
        cout << arr3[i] << " ";
        return -1;
    }
    cout << "no matching elements";
    return 0;
}

int main()
{
   int arr1[] = { 1, 2, 3, 4, 5};
   int size1 = sizeof(arr1)/sizeof(int);
   int arr2[] = { 8 , 9, 0, 1};
   int size2 = sizeof(arr2)/sizeof(int);
   intraction(arr1,size1,arr2,size2);
   
   return 0;
}
