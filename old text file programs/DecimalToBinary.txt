#include <iostream>
using namespace std;

int DecToBin(int decNum){
    int ans = 0;
    int power = 1;
    while( decNum > 0 ){
        int rem = decNum % 2;
        decNum = decNum/2;
        ans = ans+(rem*power);
        power = power*10;
    }
    return ans;
}

int main()
{
    int x;
    cout << "enter a decimal number :";
    cin >> x;
    cout << "(" << x << ")10 =" << "(" << DecToBin(x) << ")2";
    return 0;
}
